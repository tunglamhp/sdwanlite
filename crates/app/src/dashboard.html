async function refreshRib() {
  try {
    const j = await (await fetch('/api/bgp/rib')).json();
    ribData.push(j.count);
    if (ribData.length > 60) ribData.shift();
    $('rib-count').textContent = j.count;
    document.querySelector('#rib-table tbody').innerHTML = (j.routes || []).map(r =>
      '' + esc(r.prefix) + '' + esc(r.neighbor) + '' +
      '' + (r.as_path_len != null ? r.as_path_len : '—') + '' +
      '' + (r.best ? 'best' : '') + '').join('')
      || 'empty';
    drawSpark();
  } catch {}
}
let ribData = [];
function drawSpark() {
  const cv = document.getElementById('sparkline');
  const dpr = window.devicePixelRatio || 1;
  const w = cv.clientWidth || 300, h = 56;
  cv.width = w * dpr; cv.height = h * dpr;
  const ctx = cv.getContext('2d');
  ctx.scale(dpr, dpr);
  ctx.clearRect(0, 0, w, h);
  if (ribData.length < 2) return;
  const max = Math.max(...ribData, 1);
  ctx.beginPath();
  ribData.forEach((v, i) => {
    const x = (i / (ribData.length - 1)) * (w - 4) + 2;
    const y = h - 6 - (v / max) * (h - 14);
    i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
  });
  ctx.strokeStyle = getComputedStyle(document.documentElement).getPropertyValue('--accent');
  ctx.lineWidth = 2; ctx.stroke();
}

/* ---------- customizable action buttons ---------- */
function loadActions() {
  return JSON.parse(localStorage.getItem('sl-actions') || 'null') || [
    { label: 'Reload config', method: 'POST', url: '/api/reload' },
    { label: 'TLS reload', method: 'POST', url: '/api/tls/reload' },
    { label: 'Generate keypair', method: 'GET', url: '/api/mesh/keypair' }
  ];
}
function saveActions(a) { localStorage.setItem('sl-actions', JSON.stringify(a)); renderActions(); }

function renderActions() {
  const acts = loadActions();
  const bar = document.getElementById('actions-bar');
  bar.innerHTML = acts.map((a, i) =>
    '').join('')
    || 'no buttons — add one below';
  const list = document.getElementById('act-list');
  list.innerHTML = acts.map((a, i) =>
    '
' +
    '' + esc(a.label) + ' ' + a.method + ' ' + esc(a.url) + '' +
    '
').join('')
    || 'none';
}

async function runAction(i) {
  const a = loadActions()[i];
  try {
    const opt = { method: a.method };
    if (a.method !== 'GET') opt.headers = { 'Content-Type': 'application/json' };
    const r = await fetch(a.url, opt);
    const text = (await r.text()).slice(0, 200);
    alert(a.label + ' → HTTP ' + r.status + '\n' + text);
  } catch (e) { alert(a.label + ' failed: ' + e); }
}

function addAction() {
  const label = document.getElementById('act-label').value.trim();
  const method = document.getElementById('act-method').value;
  const url = document.getElementById('act-url').value.trim();
  if (!label || !url) { alert('label and URL required'); return; }
  const acts = loadActions(); acts.push({ label, method, url });
  saveActions(acts);
  document.getElementById('act-label').value = '';
  document.getElementById('act-url').value = '';
}
function delAction(i) {
  const acts = loadActions(); acts.splice(i, 1); saveActions(acts);
}
